<?php
// Generated with WP CodeKit — powered by GrowQuest (https://growquest.io).
/**
 * Plugin Name:       Product fields
 * Description:       Adds 2 fields to the Product Data metabox.
 * Version:           1.0.0
 * Requires PHP:      7.4
 * Requires Plugins:  woocommerce
 * Text Domain:       acme
 */

defined( 'ABSPATH' ) || exit;

/**
 * Render the fields in the general tab.
 */
function acme_product_fields() {
	global $product_object;

	$product = $product_object;

	woocommerce_wp_text_input(
		array(
			'id'                => 'acme_warranty_months',
			'label'             => __( 'Warranty (months)', 'acme' ),
			'description'       => __( 'Shown on the product page under the price.', 'acme' ),
			'desc_tip'          => true,
			'value'             => get_post_meta( $product->get_id(), 'acme_warranty_months', true ),
			'type'              => 'number',
			'custom_attributes' => array(
				'step' => 'any',
				'min'  => '0',
			),
		)
	);

	woocommerce_wp_checkbox(
		array(
			'id'          => 'acme_gift_wrappable',
			'label'       => __( 'Gift wrappable', 'acme' ),
			'description' => __( 'Offers gift wrapping at checkout for this product.', 'acme' ),
			'desc_tip'    => true,
			'value'       => get_post_meta( $product->get_id(), 'acme_gift_wrappable', true ) ?: 'no',
		)
	);
}
add_action( 'woocommerce_product_options_general_product_data', 'acme_product_fields' );

/**
 * Save the fields. Fires after WooCommerce has already verified the
 * Product Data metabox nonce, so no extra nonce check belongs here.
 *
 * @param int $post_id Product post ID.
 */
function acme_save_product_fields( $post_id ) {
	update_post_meta( $post_id, 'acme_warranty_months', wc_format_decimal( wp_unslash( $_POST['acme_warranty_months'] ?? '' ) ) );

	update_post_meta( $post_id, 'acme_gift_wrappable', isset( $_POST['acme_gift_wrappable'] ) ? 'yes' : 'no' );
}
add_action( 'woocommerce_process_product_meta', 'acme_save_product_fields' );

/**
 * Expose the fields to the REST API and the block editor.
 */
function acme_register_product_meta() {
	register_post_meta(
		'product',
		'acme_warranty_months',
		array(
			'type'              => 'number',
			'single'            => true,
			'show_in_rest'      => true,
			'sanitize_callback' => 'wc_format_decimal',
			'auth_callback'     => static function ( $allowed, $meta_key, $post_id ) {
				return current_user_can( 'edit_product', $post_id );
			},
		)
	);

	register_post_meta(
		'product',
		'acme_gift_wrappable',
		array(
			'type'              => 'boolean',
			'single'            => true,
			'show_in_rest'      => true,
			'sanitize_callback' => static function ( $value ) {
				return $value ? 'yes' : 'no';
			},
			'auth_callback'     => static function ( $allowed, $meta_key, $post_id ) {
				return current_user_can( 'edit_product', $post_id );
			},
		)
	);
}
add_action( 'init', 'acme_register_product_meta' );
